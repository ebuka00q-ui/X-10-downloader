(() => {
"use strict";

const $ = id => document.getElementById(id);
const audio = $("audio");

const songs = [
 {id:1,title:"Dream Never Ends",artist:"Vic-sound",categories:["rap","calm","moody"],duration:214},
 {id:2,title:"Triple Red",artist:"Vic-sound",categories:["rap","sad"],duration:198},
 {id:3,title:"Night Motion",artist:"Vic-sound",categories:["hip-hop","trap","moody"],duration:226},
 {id:4,title:"Soft Lights",artist:"Vic-sound",categories:["calm","love"],duration:201},
 {id:5,title:"New Energy",artist:"Vic-sound",categories:["pop","new"],duration:190}
];

const state = {
 view:"homeView", category:"all", currentId:null, playing:false,
 queue:songs.map(s=>s.id), repeat:new Map(), history:["homeView"],
 timer:"off", timerId:null, notifications:false, releases:[], collections:[]
};

function song(id){return songs.find(s=>s.id===id)}
function current(){return song(state.currentId)}
function initials(t){return t.split(/\s+/).slice(0,2).map(x=>x[0]).join("").toUpperCase()}
function esc(t){return String(t).replaceAll("&","&amp;").replaceAll("<","&lt;").replaceAll(">","&gt;").replaceAll('"',"&quot;").replaceAll("'","&#039;")}
function liked(id){return state.collections.some(c=>c.songIds.includes(id))}

function visibleSongs(){
 let list=songs.filter(s=>state.category==="all"||s.categories.includes(state.category));
 const q=$("searchInput").value.trim().toLowerCase();
 if(q) list=list.filter(s=>s.title.toLowerCase().includes(q)||s.artist.toLowerCase().includes(q));
 return list;
}

function renderHome(){
 $("songCount").textContent=`${songs.length} ${songs.length===1?"sound":"sounds"}`;
 const list=$("homeSongs"); list.innerHTML="";
 visibleSongs().forEach(s=>{
  const el=document.createElement("article");
  el.className="song-card"+(s.id===state.currentId?" playing":"");
  el.innerHTML=`<button class="cover song-open">${initials(s.title)}</button>
  <button class="song-details song-open"><span class="song-title">${esc(s.title)}</span><span class="song-artist">${esc(s.artist)}</span><span class="play-wave ${state.playing&&s.id===state.currentId?"":"stopped"}">〰〰〰</span></button>
  <button class="card-heart ${liked(s.id)?"liked":""}" data-like="${s.id}">${liked(s.id)?"♥":"♡"}</button>`;
  el.dataset.id=s.id; list.appendChild(el);
 });
 if(!list.children.length) list.innerHTML='<div style="text-align:center;color:var(--muted);padding:40px">No music found.</div>';
}

function renderCollections(){
 const grid=$("collections"); grid.innerHTML="";
 if(!state.collections.length){
  grid.innerHTML='<div class="collection-card"><div class="collection-covers"><div class="cover">+</div><div class="cover">+</div></div><div class="collection-title">Your collections</div><p class="collection-count">Create a collection from music.</p></div>';
  return;
 }
 state.collections.forEach(c=>{
  const cover=c.songIds.slice(0,2).map(id=>song(id)).filter(Boolean);
  const el=document.createElement("article"); el.className="collection-card";
  el.innerHTML=`<div class="collection-covers">${cover.map(s=>`<div class="cover">${initials(s.title)}</div>`).join("")}</div><div class="collection-title">${esc(c.name)}</div><p class="collection-count">${c.songIds.length} sounds</p>`;
  grid.appendChild(el);
 });
}

function renderLatest(){
 const list=$("latestList"); list.innerHTML="";
 if(!state.releases.length){list.innerHTML='<div style="text-align:center;color:var(--muted);padding-top:35vh">No release yet</div>';return}
 state.releases.forEach(r=>{
  const el=document.createElement("article"); el.className="release-card";
  el.innerHTML=`<div class="release-cover">${esc(r.name)}</div><div class="release-meta"><div><h3>${esc(r.name)}</h3><p>${esc(r.type)} · ${r.songIds.length} songs · ${esc(r.date)}</p></div><button data-release="${r.id}">${r.downloading?"Pause":"Download"}</button></div><div class="download-bar"><div class="download-progress" style="width:${r.progress}%"></div></div><div class="release-status">${esc(r.status)}</div>`;
  list.appendChild(el);
 });
}

function updateUI(){
 const s=current(); if(!s)return;
 $("miniTitle").textContent=s.title;$("miniArtist").textContent=s.artist;
 $("playerTitle").textContent=s.title;$("playerArtist").textContent=s.artist;
 $("miniCover").textContent=initials(s.title);$("playerCover").textContent=initials(s.title);
 $("miniPlay").textContent=state.playing?"Ⅱ":"▶";$("playerPlay").textContent=state.playing?"Ⅱ":"▶";
 $("miniPlayer").classList.remove("hidden");$("miniPlayer").classList.toggle("visible",true);
 $("navWave").classList.toggle("playing",state.playing);$("playerWave").classList.toggle("playing",state.playing);
 document.querySelectorAll(".play-wave").forEach(w=>w.classList.toggle("stopped",!(state.playing&&s.id===state.currentId)));
 renderQueue();
}

function start(id){
 if(!song(id))return;
 if(!state.queue.includes(id))state.queue.unshift(id);
 state.currentId=id;
 state.playing=true;
 if(song(id).src){audio.src=song(id).src;audio.play().catch(()=>{})}
 updateUI();renderHome();
}

function togglePlay(){
 if(!current()){start(state.queue[0]);return}
 if(current().src){audio.paused?audio.play().catch(()=>{}):audio.pause()}
 else{state.playing=!state.playing;updateUI();renderHome()}
}

function next(){
 if(!state.queue.length)return;
 const id=state.currentId;
 const n=state.repeat.get(id)||0;
 if(n>1){state.repeat.set(id,n-1);start(id);return}
 state.repeat.delete(id);
 let i=state.queue.indexOf(id);
 i=i<state.queue.length-1?i+1:0;
 start(state.queue[i]);
}

function previous(){
 if(audio.currentTime>4){audio.currentTime=0;return}
 let i=state.queue.indexOf(state.currentId);
 i=i>0?i-1:state.queue.length-1;
 start(state.queue[i]);
}

function renderQueue(){
 const list=$("queueList");list.innerHTML="";
 state.queue.forEach((id,i)=>{
  const s=song(id);if(!s)return;
  const n=state.repeat.get(id)||0;
  const el=document.createElement("div");el.className="queue-item"+(id===state.currentId?" playing":"");el.draggable=true;el.dataset.id=id;el.dataset.index=i;
  el.innerHTML=`<div class="queue-cover">${initials(s.title)}</div><div class="queue-title">${esc(s.title)}</div>${n>1?`<button class="repeat-count" data-repeat="${id}">×${n}</button>`:"<span></span>"}`;
  list.appendChild(el);
 });
}

function cycleRepeat(id){
 let n=state.repeat.get(id)||0;
 n=n===0?2:n<10?n+1:0;
 n?state.repeat.set(id,n):state.repeat.delete(id);
 renderQueue();
}

function reduceRepeat(id){
 const n=state.repeat.get(id)||0;
 n<=2?state.repeat.delete(id):state.repeat.set(id,n-1);
 renderQueue();
}

function openPlayer(){$("playerView").classList.remove("hidden");$("bottomNav").classList.add("hidden");updateUI()}
function closePlayer(){$("playerView").classList.add("hidden");$("bottomNav").classList.remove("hidden")}
function openQueue(){$("queueSheet").classList.remove("hidden");$("notificationBackdrop").classList.remove("hidden");renderQueue()}
function closeQueue(){$("queueSheet").classList.add("hidden");if($("timerSheet").classList.contains("hidden"))$("notificationBackdrop").classList.add("hidden")}
function openTimer(){$("timerSheet").classList.remove("hidden");$("notificationBackdrop").classList.remove("hidden")}
function closeTimer(){$("timerSheet").classList.add("hidden");if($("queueSheet").classList.contains("hidden"))$("notificationBackdrop").classList.add("hidden")}

function setTimer(mode){
 if(state.timerId)clearTimeout(state.timerId);state.timerId=null;state.timer=mode;
 document.querySelectorAll("#timerOptions button").forEach(b=>b.classList.toggle("active",b.dataset.minutes===String(mode)));
 if(mode==="off"){closeTimer();return}
 if(mode==="track"){closeTimer();return}
 state.timerId=setTimeout(()=>{audio.pause();state.playing=false;updateUI();renderHome();toast("Timer finished")},Number(mode)*60000);
 closeTimer();toast(`Timer set for ${mode} minutes`);
}

function toast(text){const t=$("toast");t.textContent=text;t.classList.add("show");clearTimeout(toast.t);toast.t=setTimeout(()=>t.classList.remove("show"),2200)}

function navigate(view){
 if(state.view===view)return;
 state.history.push(view);state.view=view;
 document.querySelectorAll(".view").forEach(v=>v.style.display=v.id===view?"block":"none");
 document.querySelectorAll(".nav-button").forEach(b=>b.classList.toggle("active",b.dataset.view===view));
 $("searchToggle").classList.toggle("hidden",view!=="homeView");
 if(view==="latestView")$("notificationDot").classList.add("hidden");
 $("notificationPanel").classList.add("hidden");
}

document.querySelectorAll(".nav-button").forEach(b=>b.addEventListener("click",()=>navigate(b.dataset.view)));

$("categories").addEventListener("click",e=>{
 const b=e.target.closest(".category");if(!b)return;
 state.category=b.dataset.category;document.querySelectorAll(".category").forEach(x=>x.classList.toggle("active",x===b));renderHome();
});

$("searchToggle").addEventListener("click",()=>{
 $("searchBox").classList.toggle("hidden");
 if(!$("searchBox").classList.contains("hidden"))$("searchInput").focus();else{$("searchInput").value="";renderHome()}
});
$("searchInput").addEventListener("input",renderHome);

$("homeSongs").addEventListener("click",e=>{
 const like=e.target.closest("[data-like]");
 if(like){e.stopPropagation();const id=Number(like.dataset.like);let c=state.collections.find(x=>x.songIds.includes(id));if(c){c.songIds=c.songIds.filter(x=>x!==id);if(!c.songIds.length)state.collections=state.collections.filter(x=>x.id!==c.id)}else{if(!state.collections.length)state.collections.push({id:Date.now(),name:"My Collection",songIds:[]});state.collections[0].songIds.push(id)}renderHome();renderCollections();return}
 const open=e.target.closest(".song-open");if(open)start(Number(open.closest(".song-card").dataset.id));
});

$("miniPlay").addEventListener("click",e=>{e.stopPropagation();togglePlay()});
$("miniPrev").addEventListener("click",e=>{e.stopPropagation();previous()});
$("miniNext").addEventListener("click",e=>{e.stopPropagation();next()});
$("miniCover").addEventListener("click",openPlayer);$("miniInfo").addEventListener("click",openPlayer);
$("playerBack").addEventListener("click",closePlayer);$("playerPlay").addEventListener("click",togglePlay);$("playerPrev").addEventListener("click",previous);$("playerNext").addEventListener("click",next);
$("queueOpen").addEventListener("click",openQueue);$("timerButton").addEventListener("click",openTimer);$("playerTimer").addEventListener("click",openTimer);
$("queueClose").addEventListener("click",closeQueue);$("timerClose").addEventListener("click",closeTimer);

$("queueList").addEventListener("dblclick",e=>{const x=e.target.closest(".queue-item");if(x)cycleRepeat(Number(x.dataset.id))});
$("queueList").addEventListener("click",e=>{const r=e.target.closest("[data-repeat]");if(r){reduceRepeat(Number(r.dataset.repeat));return}const x=e.target.closest(".queue-item");if(x)start(Number(x.dataset.id))});

let drag=null;
$("queueList").addEventListener("dragstart",e=>{const x=e.target.closest(".queue-item");if(x){drag=Number(x.dataset.index);x.style.opacity=".5"}});
$("queueList").addEventListener("dragend",e=>{const x=e.target.closest(".queue-item");if(x)x.style.opacity="";drag=null});
$("queueList").addEventListener("dragover",e=>e.preventDefault());
$("queueList").addEventListener("drop",e=>{e.preventDefault();const x=e.target.closest(".queue-item");if(!x||drag===null)return;const to=Number(x.dataset.index);const current=state.currentId;const item=state.queue.splice(drag,1)[0];state.queue.splice(to,0,item);renderQueue();state.currentId=current});

let sx=0,tx=null;
$("queueList").addEventListener("touchstart",e=>{tx=e.target.closest(".queue-item");if(tx)sx=e.changedTouches[0].clientX},{passive:true});
$("queueList").addEventListener("touchend",e=>{if(!tx)return;const dx=e.changedTouches[0].clientX-sx;if(Math.abs(dx)>80){const id=Number(tx.dataset.id);state.queue=state.queue.filter(x=>x!==id);if(id===state.currentId)state.currentId=state.queue[0]||null;renderQueue();renderHome()}tx=null},{passive:true});

$("timerOptions").addEventListener("click",e=>{const b=e.target.closest("button");if(b)setTimer(b.dataset.minutes)});
$("bellButton").addEventListener("click",e=>{e.stopPropagation();if(state.view!=="latestView")return;$("notificationPanel").classList.toggle("hidden")});
$("notificationPanel").addEventListener("click",e=>e.stopPropagation());
$("notificationBackdrop").addEventListener("click",()=>{closeQueue();closeTimer()});
$("notificationSwitch").addEventListener("change",e=>{state.notifications=e.target.checked;localStorage.setItem("vicSoundNotifications",state.notifications?"on":"off");toast(state.notifications?"Notifications on":"Notifications off")});

$("latestList").addEventListener("click",e=>{
 const b=e.target.closest("[data-release]");if(!b)return;
 const r=state.releases.find(x=>x.id===Number(b.dataset.release));if(!r)return;
 r.downloading=!r.downloading;r.status=r.downloading?"Downloading...":"Paused";renderLatest();
 if(r.downloading){
  clearInterval(r.interval);
  r.interval=setInterval(()=>{
   if(!r.downloading){clearInterval(r.interval);return}
   r.progress=Math.min(100,r.progress+5);renderLatest();
   if(r.progress>=100){
    clearInterval(r.interval);r.downloading=false;r.status="Completed";renderLatest();
    setTimeout(()=>{state.releases=state.releases.filter(x=>x.id!==r.id);renderLatest();renderHome()},1200);
   }
  },180);
 }
});

$("progress").addEventListener("input",e=>{if(audio.duration){audio.currentTime=Number(e.target.value)/100*audio.duration}});
audio.addEventListener("play",()=>{state.playing=true;updateUI();renderHome()});
audio.addEventListener("pause",()=>{state.playing=false;updateUI();renderHome()});
audio.addEventListener("timeupdate",()=>{if(audio.duration){$("progress").value=audio.currentTime/audio.duration*100;$("currentTime").textContent=format(audio.currentTime);$("duration").textContent=format(audio.duration)}});
audio.addEventListener("ended",()=>next());

function format(s){s=Math.floor(s||0);return Math.floor(s/60)+":"+String(s%60).padStart(2,"0")}

// In-app Back behavior. A normal Back closes the current overlay/player first.
// Navigation history is kept inside the PWA; the root requires a second Back.
history.replaceState({vic:"root"}, "", location.href);
history.pushState({vic:"root"}, "", location.href);

window.addEventListener("popstate",()=>{
 if(!$("playerView").classList.contains("hidden")){closePlayer();history.pushState({vic:"root"}, "", location.href);return}
 if(!$("timerSheet").classList.contains("hidden")){closeTimer();history.pushState({vic:"root"}, "", location.href);return}
 if(!$("queueSheet").classList.contains("hidden")){closeQueue();history.pushState({vic:"root"}, "", location.href);return}
 if(!$("notificationPanel").classList.contains("hidden")){$("notificationPanel").classList.add("hidden");history.pushState({vic:"root"}, "", location.href);return}
 if(state.history.length>1){state.history.pop();const v=state.history[state.history.length-1];state.view=v;document.querySelectorAll(".view").forEach(x=>x.style.display=x.id===v?"block":"none");document.querySelectorAll(".nav-button").forEach(b=>b.classList.toggle("active",b.dataset.view===v));$("searchToggle").classList.toggle("hidden",v!=="homeView");history.pushState({vic:"root"}, "", location.href);return}
 toast("Press Back again to exit Vic-sound");
});

document.addEventListener("contextmenu",e=>e.preventDefault());
document.addEventListener("selectstart",e=>{if(!["INPUT","TEXTAREA"].includes(e.target.tagName))e.preventDefault()});

document.querySelectorAll(".view").forEach(v=>v.style.display=v.id==="homeView"?"block":"none");
$("notificationSwitch").checked=localStorage.getItem("vicSoundNotifications")==="on";
state.releases=[{id:1,name:"Demo EP",type:"EP",songIds:[1,2,3],date:"2026",progress:0,downloading:false,status:"Tap to download"}];
renderHome();renderCollections();renderLatest();
})();
